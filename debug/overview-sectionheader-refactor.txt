import { useState } from "react";

import ChallengePanel from "../challenges/ChallengePanel";

export default function ChallengesSection({ locale }: { locale: Locale }) {
  const content = challengesContent[locale];

  const [activeTab, setActiveTab] = useState("challenge1");

  return (
    <section className="py-8">
      <TransitionLine text={content.transition} />

      <div className="mx-auto max-w-7xl px-6">
        {/* header */}

        <div className="max-w-4xl">
          <div className="text-[11px] font-semibold tracking-[0.18em] text-slate-400 uppercase">
            {content.sectionLabel}
          </div>

          <h2 className="mt-4 text-4xl leading-[1.02] font-bold tracking-tight text-slate-900 md:text-6xl">
            {content.title.line1}

            <br />

            {content.title.line2}
          </h2>
        </div>

        {/* tabs */}

        <div className="mt-12 inline-flex rounded-full bg-slate-100 p-2 shadow-inner">
          {content.tabs.map((tab) => (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id)}
              className={`rounded-full px-8 py-4 text-sm font-semibold transition ${
                activeTab === tab.id
                  ? "bg-slate-900 text-white shadow-sm"
                  : "text-slate-600 hover:text-slate-900"
              }`}
            >
              {tab.label}
            </button>
          ))}
        </div>

        {/* panel */}

        <div className="mt-10">
          {activeTab === "challenge1" && (
            <ChallengePanel challenge={content.challenge1} />
          )}

          {activeTab === "challenge2" && (
            <ChallengePanel challenge={content.challenge2} />
          )}

          {activeTab === "challenge3" && (
            <ChallengePanel challenge={content.challenge3} />
          )}
        </div>
      </div>
    </section>
  );
}

type Props = {
  eyebrow?: string;
  title: string;
  subtitle?: string;
  align?: "left" | "center";
  dark?: boolean;
  size?: "md" | "lg" | "xl";
};

export default function SectionHeader({
  eyebrow,
  title,
  subtitle,
  align = "left",
  dark = false,
  size = "lg",
}: Props) {
  const titleSize =
    size === "xl"
      ? "text-5xl md:text-7xl"
      : size === "lg"
        ? "text-4xl md:text-6xl"
        : "text-3xl md:text-4xl";

  return (
    <div
      className={
        align === "center" ? "mx-auto max-w-4xl text-center" : "max-w-4xl"
      }
    >
      {eyebrow && (
        <div
          className={`mb-5 font-mono text-[11px] font-bold tracking-[0.3em] uppercase ${
            dark ? "text-indigo-300" : "text-indigo-500"
          }`}
        >
          {eyebrow}
        </div>
      )}

      <h2
        className={`${titleSize} leading-[1.02] font-bold tracking-tight ${
          dark ? "text-white" : "text-slate-900"
        }`}
      >
        {title}
      </h2>

      {subtitle && (
        <p
          className={`mt-6 max-w-3xl text-lg leading-8 ${
            dark ? "text-slate-400" : "text-slate-500"
          } ${align === "center" ? "mx-auto" : ""}`}
        >
          {subtitle}
        </p>
      )}
    </div>
  );
}
